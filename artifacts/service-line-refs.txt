   138	export function knowledgeRelationsHash(relations: KnowledgeRelation[]): string {
   139		const canonical = relations
   140			.map((relation) => {
   141				// Declaration assertion IDs and selectors are verification evidence. Do not
   142				// hash arbitrary relation metadata (for example navigation annotations).
   143				const manifest = relation.metadata?.manifest;
   144				const declaration = manifest && typeof manifest === "object"
   145					? manifest as Record<string, unknown> : undefined;
   146				const evidence = declaration?.evidence;
   147				const selector = evidence && typeof evidence === "object"
   148					? (evidence as Record<string, unknown>).selector : undefined;
   149				return ({
   150				targetPath: relation.targetPath,
   151				targetKind: relation.targetKind,
   152				relationKind: relation.relationKind,
   153				provenance: relation.provenance,
   154				...(typeof declaration?.id === "string" ? { manifestId: declaration.id } : {}),
   155				...(typeof declaration?.assertionId === "string" ? { assertionId: declaration.assertionId } : {}),
   156				...(selector && typeof selector === "object" &&
   157					typeof (selector as Record<string, unknown>).kind === "string" &&
   158					typeof (selector as Record<string, unknown>).value === "string"
   159					? { selector: { kind: (selector as Record<string, string>).kind, value: (selector as Record<string, string>).value } }
   160					: {}),
   161				});
   162			})
   163			.sort((left, right) =>
   164				JSON.stringify(left).localeCompare(JSON.stringify(right)),
   165			);
   166		return sha256(JSON.stringify(canonical));
   167	}

   185		private async normalizeProjectPath(value: string): Promise<string> {
   186			const root = await realpath(this.repoRoot);
   187			const absolute = path.isAbsolute(value) ? value : path.resolve(root, value);
   188			const relative = path.relative(root, absolute);
   189			if (!relative || relative === ".") {
   190				throw new Error("Knowledge paths must point to a project file.");
   191			}
   192			if (relative.startsWith("..") || path.isAbsolute(relative)) {
   193				throw new Error(`Path escapes project root: ${value}`);
   194			}
   195			return relative.replace(/\\/g, "/");
   196		}
   197	
   198		private async resolveExistingProjectFile(filePath: string): Promise<string> {
   199			const root = await realpath(this.repoRoot);
   200			const candidate = path.join(root, filePath);
   201			const resolved = await realpath(candidate);
   202			const relative = path.relative(root, resolved);
   203			if (!relative || relative.startsWith("..") || path.isAbsolute(relative)) {
   204				throw new Error(`Path resolves outside project root: ${filePath}`);
   205			}
   206			if (!(await stat(resolved)).isFile()) {
   207				throw new Error(`Path is not a file: ${filePath}`);
   208			}
   209			return resolved;
   210		}

   213			try {

   217				return null;
   218			}
   219		}
